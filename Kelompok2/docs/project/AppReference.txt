function Icon({ name, size = 20 }: { name: string; size?: number }) {
  return <svg className="icon" width={size} height={size} viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.8" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">{iconPaths[name]}</svg>;

function Avatar({ name, small = false }: { name: string; small?: boolean }) {
  const initials = name.split(" ").map((n) => n[0]).slice(0, 2).join("");
  return <div className={`avatar ${small ? "avatar-small" : ""}`}>{initials}</div>;
}

function Repertorium({ onBack, entries, setEntries, onFeedback }: { onBack: () => void; entries: RepertoryEntry[]; setEntries: (entries: RepertoryEntry[]) => void; onFeedback: (text: string) => void }) {
  const emptyForm = { no: "", deed: "", date: "2026-03-12", type: "", party: "", nature: "Minuta", status: "Draf", notes: "" };
  const [mode, setMode] = useState<"add" | "detail" | "edit" | "period" | null>(null);
  const [selected, setSelected] = useState<RepertoryEntry | null>(null);
  const [form, setForm] = useState(emptyForm);
  const [search, setSearch] = useState("");
  const [error, setError] = useState("");
  const shown = entries.filter(e => `${e.deed} ${e.party} ${e.type}`.toLowerCase().includes(search.toLowerCase()));
  const update = (key: string, value: string) => setForm({ ...form, [key]: value });
  function openAdd() { setForm({ ...emptyForm, no: String(entries.length + 1).padStart(3, "0") }); setError(""); setMode("add"); }
  function openDetail(entry: RepertoryEntry) { setSelected(entry); setMode("detail"); }
  function openEdit() { if (!selected) return; setForm({ no: selected.no, deed: selected.deed, date: selected.date, type: selected.type, party: selected.party, nature: selected.nature, status: selected.status, notes: selected.notes }); setError(""); setMode("edit"); }
  function saveRecord() {
    if (!form.no || !form.deed || !form.date || !form.type || !form.party) return setError("Lengkapi nomor, tanggal, jenis akta, dan pihak terkait.");
    const duplicate = entries.some(e => e.deed === form.deed && e.id !== selected?.id);
    if (duplicate) return setError("Nomor akta sudah tercatat. Periksa aturan penomoran kantor.");
    if (mode === "edit" && selected) {
      const corrected = { ...selected, ...form, corrected: true };
      setEntries(entries.map(e => e.id === selected.id ? corrected : e));
      setSelected(corrected);
      setMode("detail");
      onFeedback("Koreksi repertorium dicatat dalam riwayat");
    } else {
      setEntries([...entries, { ...form, id: `REP-${Date.now()}` }]);
      setMode(null);
      onFeedback("Entri repertorium berhasil ditambahkan");
    }
  }
  return <div className="page"><AppBar title="Repertorium" onBack={onBack} action={<button className="icon-button" onClick={openAdd} aria-label="Tambah repertorium"><Icon name="plus" /></button>} /><main className="content"><div className="search"><Icon name="search" /><input value={search} onChange={e => setSearch(e.target.value)} placeholder="Cari nomor akta atau pihak terkait" /></div><div className="list-heading"><span>{shown.length} catatan · Maret 2026</span><button onClick={() => setMode("period")}><Icon name="calendar" size={15} /> Periode</button></div><div className="record-list">{shown.map(e => <button key={e.id} className="record-card" onClick={() => openDetail(e)}><span className="record-number">{e.no}</span><div><div><strong>Akta No. {e.deed}</strong><Badge tone={e.status === "Terbit" ? "success" : "warning"}>{e.status}</Badge></div><p>{e.type}</p><span>{e.party}</span><small><Icon name="calendar" size={13} /> {new Date(e.date).toLocaleDateString("id-ID", { day: "2-digit", month: "short", year: "numeric" })}{e.corrected ? " · Dikoreksi" : ""}</small></div><Icon name="chevron" /></button>)}</div>{shown.length === 0 && <div className="empty"><div><Icon name="search" /></div><strong>Catatan tidak ditemukan</strong><p>Periksa kata kunci pencarian.</p></div>}<div className="info-note"><Icon name="file" /><span>Repertorium menyimpan data terstruktur, bukan dokumen atau lampiran. Nomor akta tidak dibuat otomatis.</span></div></main>
  {(mode === "add" || mode === "edit") && <Modal title={mode === "edit" ?   {mode === "period" && <Modal title="Filter Periode" onClose={() => setMode(null)}><div className="form-stack"><label>Rentang tanggal<input type="month" defaultValue="2026-03" /></label><button className="primary-button" onClick={() => setMode(null)}>Terapkan Periode</button></div></Modal>}</div>;

function Staff({ onBack, onFeedback }: { onBack: () => void; onFeedback: (text: string) => void }) {
  const [staff, setStaff] = useState([{ name: "Andi Pratama", email: "andi@kantor.id", active: true }, { name: "Rina Amelia", email: "rina@kantor.id", active: true }, { name: "Fajar Nugroho", email: "fajar@kantor.id", active: false }]);
  const [editing, setEditing] = useState<number | "new" | null>(null);
  const [name, setName] = useState("");
  const [email, setEmail] = useState("");
  function openEditor(index: number | "new") { setEditing(index); setName(index === "new" ? "" : staff[index].name); setEmail(index === "new" ? "" : staff[index].email); }
  function save() { if (!name || !email.includes("@")) return; if (editing === "new") setStaff([...staff, { name, email, active: true }]); else if (typeof editing === "number") setStaff(staff.map((s, i) => i === editing ? { ...s, name, email } : s)); setEditing(null); onFeedback("Akun staf berhasil disimpan"); }
  function toggle(index: number) { setStaff(staff.map((s, i) => i === index ? { ...s, active: !s.active } : s)); onFeedback(`Akun ${staff[index].active ? "dinonaktifkan" : "diaktifkan"}`); }
  return <div className="page"><AppBar title="Manajemen Staf" onBack={onBack} action={<button className="icon-button" onClick={() => openEditor("new")} aria-label="Tambah staf"><Icon name="plus" /></button>} /><main className="content"><div className="search"><Icon name="search" /><input placeholder="Cari nama atau email staf" /></div><div className="staff-list">{staff.map((person, i) => <div className="staff-card" key={person.email}><Avatar name={person.name} /><div><strong>{person.name}</strong><p>{person.email}</p><Badge tone={person.active ? "success" : "neutral"}>{person.active ? "Aktif" : "Nonaktif"}</Badge></div><button className="icon-button" onClick={() => openEditor(i)} aria-label={`Edit ${person.name}`}><Icon name="more" /></button></div>)}</div></main>{editing !== null && <Modal title={editing === "new" ? "Tambah Staf" : "Edit Staf"} onClose={() => setEditing(null)}><div className="form-stack"><label>Nama staf<input value={name} onChange={e => setName(e.target.value)} /></label><label>Email<input value={email} onChange={e => setEmail(e.target.value)} /></label><label>Peran<select><option>Staf</option></select></label>{typeof editing === "number" && <button className="secondary-button" onClick={() => { toggle(editing); setEditing(null); }}>{staff[editing].active ? "Nonaktifkan Akun" : "Aktifkan Akun"}</button>}<button className="primary-button" disabled={!name || !email.includes("@")} onClick={save}>Simpan Akun</button></div></Modal>}</div>;
}

export default function App() {
  const [splash, setSplash] = useState(() => !new URLSearchParams(window.location.search).has("app"));
  const [loggedIn, setLoggedIn] = useState(false);
  const [page, setPage] = useState<Page>("home");
  const [role, setRole] = useState<Role>("Notaris");
  const [services, setServices] = useState(initialServices);
  const [selected, setSelected] = useState<Service>(initialServices[1]);
  const [serviceFilter, setServiceFilter] = useState("Semua");
  const [notes, setNotes] = useState<Record<string, ServiceNote[]>>(initialNotes);
  const [activities, setActivities] = useState<Record<string, Activity[]>>(initialActivities);
  const [repertory, setRepertory] = useState<RepertoryEntry[]>(initialRepertory);
  const [emptyCategoryConfirmed, setEmptyCategoryConfirmed] = useState(false);
  const [logoutConfirm, setLogoutConfirm] = useState(false);
  const [toast, setToast] = useState("");
  const rootPages = useMemo(() => ["home", "services", "add", "reports", "profile"].includes(page), [page]);
  function navigate(p: Page) { setPage(p); window.scrollTo(0, 0); }
  function feedback(text: string) { setToast(text); window.setTimeout(() => setToast(""), 2400); }
  function openFiltered(filter: string) { setServiceFilter(filter); navigate("services"); }
  function openService(s: Service) { setSelected(s); navigate("detail"); }
  function updateService(s: Service) { setSelected(s); setServices(prev => prev.map(x => x.id === s.id ? s : x)); feedback("Perubahan berhasil disimpan"); }
  function addNote(note: ServiceNote) { setNotes(prev => ({ ...prev, [selected.id]: [note, ...(prev[selected.id] || [])] })); feedback("Catatan berhasil disimpan"); }
  function addActivity(activity: Activity) { setActivities(prev => ({ ...prev, [selected.id]: [activity, ...(prev[selected.id] || [])] })); }
  function saveService(s: Service) {
    setServices(prev => [s, ...prev]);
    setNotes(prev => ({ ...prev, [s.id]: [] }));
    setActivities(prev => ({ ...prev, [s.id]: [{ id: `A-${Date.now()}`, title: "Layanan dibuat", text: `${role === "Notaris" ? "Nadia Larasati" : "Andi Pratama"} membuat layanan dan menetapkan tahap Penerimaan Berkas.`, time: "Baru saja", tone: "green" }] }));
    setSelected(s);
    feedback(`${s.id} berhasil dibuat`);
    navigate("detail");
  }
  if (splash) return <div className="phone-shell" onClick={() => setSplash(false)}><Splash /></div>;
  if (!loggedIn) return <div className="phone-shell"><Login onLogin={() => setLoggedIn(true)} /></div>;
  let screen: ReactNode;
  if (page === "home") screen = <Home services={services} onOpen={openService} navigate={navigate} openFiltered={openFiltered} />;
  else if (page === "services") screen = <Services services={services} onOpen={openService} navigate={navigate} initialFilter={serviceFilter} onFilterChange={setServiceFilter} />;
  else if (page === "add") screen = <AddService onSave={saveService} onBack={() => navigate("home")} />;
  else if (page === "detail") screen = <Detail service={selected} onBack={() => navigate("services")} onUpdate={updateService} role={role} notes={notes[selected.id] || []} activities={activities[selected.id] || []} onAddNote={addNote} onAddActivity={addActivity} />;
  else if (page === "reports") screen = <Reports navigate={navigate} role={role} emptyConfirmed={emptyCategoryConfirmed} setEmptyConfirmed={setEmptyCategoryConfirmed} recordCount={repertory.length} />;
  else if (page === "reportPreview") screen = <ReportPreview onBack={() => navigate("reports")} role={role} onFeedback={feedback} />;
  else if (page === "repertorium") screen = <Repertorium onBack={() => navigate("reports")} entries={repertory} setEntries={setRepertory} onFeedback={feedback} />;
  else if (page === "notifications") screen = <Notifications onBack={() => navigate("home")} onOpen={() => openService(services[2])} />;
  else if (page === "staff") screen = <Staff onBack={() => navigate("profile")} onFeedback={feedback} />;
  else screen = <Profile role={role} setRole={setRole} navigate={navigate} onLogout={() => setLogoutConfirm(true)} onFeedback={feedback} />;
  return <div className="app-canvas"><div className="phone-shell">{screen}{rootPages && <BottomNav page={page} navigate={navigate} />}{toast && <div className="toast"><Icon name="check" />{toast}</div>}{logoutConfirm && <div className="dialog-overlay"><div className="dialog"><div className="dialog-icon"><Icon name="logout" /></div><h2>Keluar dari akun?</h2><p>Anda perlu masuk kembali untuk mengakses layanan kantor.</p><div className="button-row"><button className="secondary-button" onClick={() => setLogoutConfirm(false)}>Batal</button><button className="danger-button" onClick={() => { setLogoutConfirm(false); setLoggedIn(false); setPage("home"); }}>Ya, Keluar</button></div></div></div>}</div></div>;
}
